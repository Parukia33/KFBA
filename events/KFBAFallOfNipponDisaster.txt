	id = kfba_fall_of_nippon_disaster.3
	title = kfba_fall_of_nippon_disaster.3.t
	desc = kfba_fall_of_nippon_disaster.3.d
	picture = CIVIL_WAR_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	major = yes
	
	trigger = {
		NOT = { exists = ABC }
		is_year = 1446
	}
	
	immediate = {
		hidden_effect = {
			214 = {
				add_core = ABC
			}
			221 = {
				add_core = ABC
			}
			226 = {
				add_core = ABC
			}
			214 = {
				add_core = ABC
			}
			if = {
				limit = {
					213 = {
						owned_by = ROOT
					}
				}
				213 = {
					add_core = ABC
				}
			}
			if = {
				limit = {
					212 = {
						owned_by = ROOT
					}
				}
				212 = {
					add_core = ABC
				}
			}
			if = {
				limit = {
					211 = {
						owned_by = ROOT
					}
				}
				211 = {
					add_core = ABC
				}
			}
			release = ABC
			ABC = {
				change_religion = arceusism
				change_government = monarchy
				add_government_reform = daimyo
			}
			add_truce_with = ABC	
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.3.a
		vassalize = ABC
		every_owned_province = {
			limit = {
				region = nippon_region
			}
			add_claim = ABC
		}
		hidden_effect = {
			ABC = {
				capital_scope = {
					build_to_forcelimit = {
						infantry = 0.6
						cavalry = 0.3
						artillery = 0.1
					}
				}
				random_owned_province = {
					limit = {
						has_port = yes
					}
					build_to_forcelimit = {
						galley = 0.5
						light_ship = 0.2
						transport = 0.2
					}
				}
			}
		}
	}
}

# Pretender to Nippon
country_event = {
	id = kfba_fall_of_nippon_disaster.5
	title = kfba_fall_of_nippon_disaster.5.title
	desc = kfba_fall_of_nippon_disaster.5.desc
	picture = NUSANTARA_THRONE_eventPicture
	
	is_triggered_only = yes
	
	trigger = {
		has_disaster = kfba_fall_of_nippon_disaster
		NOT = { has_country_flag = alp_no_more_pretenders }
		any_owned_province = {
			region = nippon_region
			controlled_by = ROOT
		}
	}
	immediate = {
		hidden_effect = {
			set_saved_name = {
				key = alp_pretender
				type = leader
			}
			random_owned_province = {
				limit = {
					region = nippon_region
					controlled_by = ROOT
				}
				save_event_target_as = alp_pretender_province
				spawn_rebels = {
					type = pretender_rebels
					leader = alp_pretender
					size = 2
				}
			}
			set_country_flag = alp_pretender_event
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.5.a
		tooltip = {
			event_target:alp_pretender_province = {
				spawn_rebels = {
					type = pretender_rebels
					leader = alp_pretender
					size = 2
				}
			}
		}
	}
}
