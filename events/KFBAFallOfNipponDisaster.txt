# The Rise of Demak
country_event = {
	id = kfba_fall_of_nippon_disaster.3
	title = kfba_fall_of_nippon_disaster.3.t
	desc = kfba_fall_of_nippon_disaster.3.d
	picture = CIVIL_WAR_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	major = yes
	
	trigger = {
		NOT = { has_country_flag = maj_no_more_bad_religion_events }
		NOT = { exists = DMK }
		NOT = { legitimacy = 75 }
		629 = {
			religion = sunni
			owned_by = ROOT
		}
		num_of_owned_provinces_with = {
			value = 4
			religion = sunni
			region = indonesia_region
		}
	}
	
	immediate = {
		hidden_effect = {
			DMK = {
				set_saved_name = {
					key = demak_sultan
					type = leader
				}
			}
			629 = {
				add_core = DMK
			}
			if = {
				limit = {
					4792 = {
						owned_by = ROOT
					}
				}
				4792 = {
					add_core = DMK
				}
			}
			if = {
				limit = {
					4793 = {
						owned_by = ROOT
					}
				}
				4793 = {
					add_core = DMK
				}
			}
			if = {
				limit = {
					2689 = {
						owned_by = ROOT
					}
				}
				2689 = {
					add_core = DMK
				}
			}
			release = DMK
			DMK = {
				change_religion = sunni
				change_government = monarchy
				add_government_reform = mandala_reform
				define_ruler = {
					name = demak_sultan
					dynasty = ROOT
					claim = 95
				}
			}
			add_truce_with = DMK	
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.3.a
		every_owned_province = {
			limit = {
				region = indonesia_region
			}
			add_claim = DMK
		}
		hidden_effect = {
			DMK = {
				capital_scope = {
					build_to_forcelimit = {
						infantry = 0.6
						cavalry = 0.3
						artillery = 0.1
					}
				}
				random_owned_province = {
					limit = {
						has_port = yes
					}
					build_to_forcelimit = {
						galley = 0.5
						light_ship = 0.2
						transport = 0.2
					}
				}
			}
		}
	}
}

# Subject liberty desire
country_event = {
	id = kfba_fall_of_nippon_disaster.5
	title = kfba_fall_of_nippon_disaster.5.title
	desc = kfba_fall_of_nippon_disaster.5.desc
	picture = {
		trigger = { has_dlc = "Star and Crescent" }
		picture = CONQUEST_eventPicture 
	}
	picture = {
		trigger = { NOT = { has_dlc = "Star and Crescent" } }
		picture = TRADEGOODS_eventPicture
	}
	
	is_triggered_only = yes
	
	trigger = {
		NOT = { has_country_flag = maj_no_more_subject_liberty }
		any_subject_country = {
			liberty_desire = 50
			ai = yes
			NOT = { truce_with = ROOT }
			NOT = { ROOT = { truce_with = PREV } }
		}
	}
	
	immediate = {
		hidden_effect = {
			random_subject_country = {
				limit = {
					is_subject_of_type = tributary_state
					liberty_desire = 50
					ai = yes
				}
				save_event_target_as = maj_tributary_country
			}
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.5.a
		ai_chance = {
			factor = 1
		}
		if = {
			limit = {
				event_target:maj_tributary_country = {
					liberty_desire = 75
				}
			}
			event_target:maj_tributary_country = {
				grant_independence = yes
			}
		}
		else = {
			event_target:maj_tributary_country = {
				add_liberty_desire = 15
			}
		}
	}
	option = {
		name = kfba_fall_of_nippon_disaster.7.b
		trigger = {
			event_target:maj_tributary_country = {
				NOT = { liberty_desire = 80 }
			}
		}
		ai_chance = {
			factor = 0
		}
		add_legitimacy = -10
		event_target:maj_tributary_country = {
			add_liberty_desire = -5
		}
	}
}

# Spread of Islam in [province]
country_event = {
	id = kfba_fall_of_nippon_disaster.8
	title = kfba_fall_of_nippon_disaster.8.title
	desc = kfba_fall_of_nippon_disaster.8.desc
	picture = {
		trigger = { has_dlc = "Star and Crescent" }
		picture = PIETY_eventPicture
	}
	picture = {
		trigger = { NOT = { has_dlc = "Star and Crescent" } }
		picture = RELIGION_eventPicture
	}
	
	is_triggered_only = yes
	
	trigger = {
		OR = {
			religion = sunni
			NOT = { has_country_flag = maj_no_more_bad_religion_events }
		}
		has_country_flag = maj_accepted_muslim_help
		any_owned_province = {
			region = indonesia_region
			NOT = { religion = sunni }
			OR = {
				has_port = yes
				any_neighbor_province = {
					religion = sunni
				}
			}
		}
	}
	
	immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					region = indonesia_region
					NOT = { religion = sunni }
					OR = {
						has_port = yes
						any_neighbor_province = {
							religion = sunni
						}
					}
				}
				save_event_target_as = maj_sunni_province
			}
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.8.a
		event_target:maj_sunni_province = {
			change_religion = sunni
		}
	}
}
