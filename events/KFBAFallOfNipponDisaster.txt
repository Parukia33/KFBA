# The Rise of Demak
country_event = {
	id = kfba_fall_of_nippon_disaster.3
	title = kfba_fall_of_nippon_disaster.3.t
	desc = kfba_fall_of_nippon_disaster.3.d
	picture = CIVIL_WAR_eventPicture
	
	fire_only_once = yes
	is_triggered_only = yes
	major = yes
	
	trigger = {
		NOT = { has_country_flag = maj_no_more_bad_religion_events }
		NOT = { exists = DMK }
		NOT = { legitimacy = 75 }
		629 = {
			religion = sunni
			owned_by = ROOT
		}
		num_of_owned_provinces_with = {
			value = 4
			religion = sunni
			region = indonesia_region
		}
	}
	
	immediate = {
		hidden_effect = {
			DMK = {
				set_saved_name = {
					key = demak_sultan
					type = leader
				}
			}
			629 = {
				add_core = DMK
			}
			if = {
				limit = {
					4792 = {
						owned_by = ROOT
					}
				}
				4792 = {
					add_core = DMK
				}
			}
			if = {
				limit = {
					4793 = {
						owned_by = ROOT
					}
				}
				4793 = {
					add_core = DMK
				}
			}
			if = {
				limit = {
					2689 = {
						owned_by = ROOT
					}
				}
				2689 = {
					add_core = DMK
				}
			}
			release = DMK
			DMK = {
				change_religion = sunni
				change_government = monarchy
				add_government_reform = mandala_reform
				define_ruler = {
					name = demak_sultan
					dynasty = ROOT
					claim = 95
				}
			}
			add_truce_with = DMK	
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.3.a
		every_owned_province = {
			limit = {
				region = indonesia_region
			}
			add_claim = DMK
		}
		hidden_effect = {
			DMK = {
				capital_scope = {
					build_to_forcelimit = {
						infantry = 0.6
						cavalry = 0.3
						artillery = 0.1
					}
				}
				random_owned_province = {
					limit = {
						has_port = yes
					}
					build_to_forcelimit = {
						galley = 0.5
						light_ship = 0.2
						transport = 0.2
					}
				}
			}
		}
	}
}

# Pretender to Majapahit
country_event = {
	id = kfba_fall_of_nippon_disaster.5
	title = kfba_fall_of_nippon_disaster.5.title
	desc = kfba_fall_of_nippon_disaster.5.desc
	picture = NUSANTARA_THRONE_eventPicture
	
	is_triggered_only = yes
	
	trigger = {
		has_disaster = kfba_fall_of_nippon_disaster
		NOT = { has_country_flag = maj_no_more_pretenders }
		any_owned_province = {
			region = indonesia_region
			controlled_by = ROOT
		}
	}
	immediate = {
		hidden_effect = {
			set_saved_name = {
				key = maj_pretender
				type = leader
			}
			random_owned_province = {
				limit = {
					region = indonesia_region
					controlled_by = ROOT
				}
				save_event_target_as = maj_pretender_province
				spawn_rebels = {
					type = pretender_rebels
					leader = maj_pretender
					size = 2
				}
			}
			set_country_flag = maj_pretender_event
		}
	}
	
	option = {
		name = kfba_fall_of_nippon_disaster.5.a
		tooltip = {
			event_target:maj_pretender_province = {
				spawn_rebels = {
					type = pretender_rebels
					leader = maj_pretender
					size = 2
				}
			}
		}
	}
}
