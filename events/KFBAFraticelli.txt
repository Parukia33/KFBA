country_event = {
    id = kfba_fraticelli.1
    title = "kfba_fraticelli.1.title"
    desc = "kfba_fraticelli.1.desc"
    picture = "LUTHER_eventPicture"

    major = yes
	fire_only_once = yes

    trigger = {
		tag = LAN
        NOT = { is_religion_enabled = fraticelli }
		is_defender_of_faith = no
        50 = { owned_by = ROOT }
		has_global_flag = kfba_the_pope_is_gone_flag
    }

    mean_time_to_happen = {
		months = 12
	}
    
	#Heretics!
    option = {
        name = "kfba_fraticelli.1.a"
		ai_chance = {
			factor = 1
		}
        enable_religion = fraticelli

			change_religion = fraticelli
			add_reform_center = fraticelli
			add_permanent_province_modifier = {
				name = "religious_zeal_at_conv"
				duration = 9000
			}
			add_province_triggered_modifier = religious_center
		}
    }
	#They see the light!
    option = {
        name = "kfba_fraticelli.1.b"
		ai_chance = {
			factor = 0
		}
        enable_religion = fraticelli
		change_religion = fraticelli
		50 = {
			change_religion = fraticelli
			add_reform_center = fraticelli
			add_permanent_province_modifier = {
				name = "religious_zeal_at_conv"
				duration = 9000
			}
			add_province_triggered_modifier = religious_center
		}
		# add_papal_influence = 100
		capital_scope = {
			change_religion = fraticelli
			add_cardinal = yes
		}
    }
}
